cmake_minimum_required(VERSION 3.20)
project(NetworkProgramming LANGUAGES C)
if(NOT UNIX)
    message(FATAL_ERROR "These POSIX examples support Linux and macOS; use WSL on Windows")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(ENABLE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(ENABLE_TLS "Build the OpenSSL TLS client" ON)
option(ENABLE_LIBEVENT "Build both libevent servers" ON)

add_library(net_options INTERFACE)
target_include_directories(net_options INTERFACE "${PROJECT_SOURCE_DIR}")
target_compile_definitions(net_options INTERFACE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
if(APPLE)
    target_compile_definitions(net_options INTERFACE _DARWIN_C_SOURCE)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(net_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
    if(ENABLE_SANITIZERS)
        target_compile_options(net_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(net_options INTERFACE -fsanitize=address,undefined)
    endif()
endif()
find_package(Threads REQUIRED)
add_library(net_core STATIC common/net.c common/echo.c common/stream.c common/select_server.c
    common/operator.c DieWithMessage.c TCPClientUtility.c TCPServerUtility.c)
target_link_libraries(net_core PUBLIC net_options)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(net_core PRIVATE common/epoll_server.c)
endif()

foreach(encoding bin text)
    if(encoding STREQUAL "bin")
        set(source VoteEncodingBin.c)
    else()
        set(source VoteEncodingText.c)
    endif()
    add_library(codec-${encoding} STATIC ${source})
    target_link_libraries(codec-${encoding} PUBLIC net_options)
endforeach()
add_library(framer-length STATIC LengthFramer.c)
add_library(framer-delim STATIC DelimFramer.c)
target_link_libraries(framer-length PUBLIC net_options)
target_link_libraries(framer-delim PUBLIC net_options)

if(ENABLE_LIBEVENT)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBEVENT REQUIRED IMPORTED_TARGET libevent>=2.1)
    add_library(net_event STATIC common/libevent_server.c)
    target_link_libraries(net_event PUBLIC net_core PkgConfig::LIBEVENT)
endif()
if(ENABLE_TLS)
    find_package(OpenSSL 1.1.1 REQUIRED)
endif()

file(READ "${PROJECT_SOURCE_DIR}/examples.json" examples)
string(JSON count LENGTH "${examples}")
math(EXPR last "${count} - 1")
foreach(index RANGE ${last})
    string(JSON item GET "${examples}" ${index})
    foreach(field name source feature platform model)
        string(JSON ${field} GET "${item}" ${field})
    endforeach()
    if(platform STREQUAL "linux" AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        continue()
    endif()
    if((feature STREQUAL "libevent" AND NOT ENABLE_LIBEVENT) OR (feature STREQUAL "tls" AND NOT ENABLE_TLS))
        continue()
    endif()
    add_executable(${name} ${source})
    string(JSON library_count LENGTH "${item}" libraries)
    math(EXPR library_last "${library_count} - 1")
    foreach(library_index RANGE ${library_last})
        string(JSON library GET "${item}" libraries ${library_index})
        target_link_libraries(${name} PRIVATE ${library})
    endforeach()
    if(model STREQUAL "server" OR model MATCHES "^multiplex-")
        list(APPEND echo_targets ${name})
    endif()
    set(model_${name} "${model}")
endforeach()

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    foreach(kind length delim)
        add_executable(test-framer-${kind} tests/test_framers.c)
        target_link_libraries(test-framer-${kind} PRIVATE framer-${kind})
        if(kind STREQUAL "length")
            target_compile_definitions(test-framer-${kind} PRIVATE LENGTH_FRAMER)
        endif()
        add_test(NAME framing-${kind} COMMAND test-framer-${kind})
    endforeach()
    foreach(kind bin text)
        add_executable(test-codec-${kind} tests/test_codecs.c)
        target_link_libraries(test-codec-${kind} PRIVATE codec-${kind})
        if(kind STREQUAL "bin")
            target_compile_definitions(test-codec-${kind} PRIVATE BINARY_CODEC)
        endif()
        add_test(NAME codec-${kind} COMMAND test-codec-${kind})
    endforeach()
    add_executable(test-net tests/test_net.c)
    target_link_libraries(test-net PRIVATE net_core Threads::Threads)
    add_test(NAME socket-helpers COMMAND test-net)
    add_executable(test-operator tests/test_operator.c)
    target_link_libraries(test-operator PRIVATE net_core)
    add_test(NAME arithmetic-codec COMMAND test-operator)
    foreach(name IN LISTS echo_targets)
        set(arguments --case echo --server ${name})
        if(model_${name} STREQUAL "multiplex-rot13")
            list(APPEND arguments --rot13)
        endif()
        if(model_${name} MATCHES "^multiplex-")
            list(APPEND arguments --multiplex)
        endif()
        add_test(NAME network-${name} COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/integration.py"
            --bin-dir "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}" ${arguments})
    endforeach()
    foreach(case clients udp votes chat arithmetic)
        add_test(NAME network-${case} COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/integration.py"
            --bin-dir "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}" --case ${case})
    endforeach()
    if(ENABLE_TLS)
        find_program(OPENSSL_COMMAND NAMES openssl REQUIRED)
        add_test(NAME network-tls COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/integration.py"
            --bin-dir "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}" --case tls --openssl "${OPENSSL_COMMAND}")
    endif()
    add_test(NAME documentation COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tools/catalog.py" --check)
    get_property(all_tests DIRECTORY PROPERTY TESTS)
    set_tests_properties(${all_tests} PROPERTIES TIMEOUT 60
        FAIL_REGULAR_EXPRESSION "AddressSanitizer;UndefinedBehaviorSanitizer;runtime error:;LeakSanitizer")
endif()
